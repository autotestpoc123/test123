import { Injectable } from '@angular/core';
import * as XLSX from 'xlsx';
import * as FileSaver from 'file-saver';
import { DatePipe } from '@angular/common';

const EXCEL_TYPE = 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet;charset=UTF-8';
const EXCEL_EXTENSION = '.xlsx';

@Injectable({
  providedIn: 'root'
})
export class DownloadExcelService {

  constructor(private datePipe: DatePipe) { }

  exportToExcel(json:any[], excelfileName:string, headers?:string[]): void {
    const flattened = (json || []).map(row => this.flattenRow(row));
    const worksheet: XLSX.WorkSheet = XLSX.utils.json_to_sheet(flattened, {header: headers});
    this.neutralizeFormulas(worksheet);
    const workbook: XLSX.WorkBook = { Sheets : {'data':worksheet}, SheetNames:['data'] };
    const excelBuffer: any = XLSX.write(workbook, {bookType: 'xlsx', type:'array'});
    this.saveAsExcelFile(excelBuffer, excelfileName);
  }

  // A worksheet cell can only hold a scalar. Array-valued fields — e.g. several
  // PersonalMobilePhone numbers — are otherwise dropped by json_to_sheet and the
  // data is silently lost. Join them into one comma-separated string so every
  // value lands in a single cell (matching how the UI renders multi-value fields).
  private flattenRow(row:any): any {
    if (row === null || typeof row !== 'object' || Array.isArray(row)) {
      return row;
    }
    const flat:any = {};
    for (const key of Object.keys(row)) {
      const value = row[key];
      flat[key] = Array.isArray(value) ? value.join(', ') : value;
    }
    return flat;
  }

  // Defend against CSV/Excel formula injection (CWE-1236) WITHOUT mutating the
  // exported values. In the .xlsx format a formula must be declared with an
  // explicit <f> element; a value beginning with = + - @ that is stored as an
  // explicit string cell ('s') is kept and displayed as literal text and is never
  // evaluated when the workbook is opened. Forcing the string type (and stripping
  // any formula binding) therefore neutralizes injection without adding a leading
  // apostrophe, so exported columns keep their original data intact — phone
  // numbers stay "+8613..." rather than becoming "'+8613...".
  private neutralizeFormulas(worksheet: XLSX.WorkSheet): void {
    for (const address of Object.keys(worksheet)) {
      if (address.charAt(0) === '!') { continue; } // skip !ref, !cols, ... metadata keys
      const cell = worksheet[address] as XLSX.CellObject;
      if (cell && typeof cell.v === 'string') {
        cell.t = 's';
        if ('f' in cell) { delete (cell as any).f; }
      }
    }
  }

  saveAsExcelFile(buffer:any, fileName:string): void{
    const data: Blob = new Blob([buffer], {type: EXCEL_TYPE});
    let now = new Date();
    FileSaver.saveAs(data, fileName + '_' + this.datePipe.transform(new Date(), 'yyyyMMdd') + EXCEL_EXTENSION);
  }
}
